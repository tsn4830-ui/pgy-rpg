/* 對話打字機：字一個一個浮出來；點對話框可直接顯示全文 */
function typewrite(){
  if(window.matchMedia && matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  const boxes = [...V().querySelectorAll(".says")]; if(!boxes.length) return;
  let t = 0;
  boxes.forEach(box => {
    const walker = document.createTreeWalker(box, NodeFilter.SHOW_TEXT, {acceptNode: n => n.parentElement.closest(".who") || !n.nodeValue.trim() && !n.nodeValue.includes("\n") ? NodeFilter.FILTER_REJECT : NodeFilter.FILTER_ACCEPT});
    const nodes = []; while(walker.nextNode()) nodes.push(walker.currentNode);
    const total = nodes.reduce((a,n) => a + n.nodeValue.length, 0); if(!total) return;
    const per = Math.min(26, 1500 / total);
    nodes.forEach(n => {
      const f = document.createDocumentFragment();
      for(const ch of n.nodeValue){ const s = document.createElement("span"); s.className = "tw"; s.textContent = ch; s.style.animationDelay = (t += per).toFixed(0) + "ms"; f.appendChild(s); }
      n.parentNode.replaceChild(f, n);
    });
    t += 120;
  });
  const v = V(); v.classList.add("typing");
  const done = () => { v.classList.remove("typing"); v.removeEventListener("click", skip, true); };
  const skip = e => { if(e.target.closest(".says")) done(); };
  v.addEventListener("click", skip, true); clearTimeout(typewrite.t); typewrite.t = setTimeout(done, t + 200);
}
